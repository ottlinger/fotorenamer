15  package de.aikiit.fotorenamer.cli;
16  
17  import de.aikiit.fotorenamer.exception.InvalidDirectoryException;
18  import de.aikiit.fotorenamer.exception.NoFilesFoundException;
19  import de.aikiit.fotorenamer.image.CreationDateFromExifImageRenamer;
20  import de.aikiit.fotorenamer.image.ImageRenamer;
21  import de.aikiit.fotorenamer.image.RemoveExifPrefixRenamer;
22  import de.aikiit.fotorenamer.image.RenameResult;
23  import de.aikiit.fotorenamer.util.Version;
24  import picocli.CommandLine;
25  import picocli.CommandLine.Command;
26  import picocli.CommandLine.Option;
27  import picocli.CommandLine.Parameters;
28  import picocli.CommandLine.Spec;
29  import picocli.CommandLine.Model.CommandSpec;
30  
31  import java.nio.file.Files;
32  import java.nio.file.Path;
33  import java.util.concurrent.Callable;
34  
35  /** Headless console entry point sharing the GUI's image processing (issue #15). */
36  @Command(name = "fotorenamer", mixinStandardHelpOptions = true,
37          version = "FotoRenamer " + Version.VERSION,
38          description = "Rename images using their EXIF capture timestamp.",
39          footer = "Back up your images before running this tool. Only the selected directory is processed.")
40  public final class ConsoleRunner implements Callable<Integer> {
41  
42      @Parameters(index = "0", paramLabel = "DIRECTORY", description = "Directory containing the images.")
43      private Path directory;
44  
45      @Option(names = {"-r", "--reverse"}, description = "Remove one leading EXIF timestamp prefix instead of adding it.")
46      private boolean reverse;
47  
48      @Spec
49      private CommandSpec spec;
50  
51      /** Launches the CLI and returns its result to the operating system. */
52      public static void main(final String[] args) {
53          System.exit(new CommandLine(new ConsoleRunner()).execute(args));
54      }
55  
56      /** Processes images synchronously without initializing Swing. */
57      @Override
58      public Integer call() {
59          if (!Files.isDirectory(directory)) {
60              throw new CommandLine.ParameterException(spec.commandLine(), "Not a directory: " + directory);
61          }
62          ImageRenamer renamer;
63          try {
64              renamer = reverse
65                      ? new RemoveExifPrefixRenamer(directory.toString())
66                      : new CreationDateFromExifImageRenamer(directory.toString());
67          } catch (NoFilesFoundException e) {
68              spec.commandLine().getErr().println("No images found in: " + directory.toAbsolutePath());
69              return 1;
70          } catch (InvalidDirectoryException e) {
71              spec.commandLine().getErr().println("Unable to read directory: " + directory.toAbsolutePath());
72              return 1;
73          }
74          RenameResult result = renamer.renameFiles();
75          spec.commandLine().getOut().printf(
76                  "Directory: %s%nProcessed: %d / %d%nRenamed: %d%nSkipped: %d%nFailed: %d%n",
77                  directory.toAbsolutePath(), result.processed(), result.total(),
78                  result.renamed(), result.skipped(), result.failed());
79          return result.failed() == 0 && !result.cancelled() ? 0 : 1;
80      }
81  }
