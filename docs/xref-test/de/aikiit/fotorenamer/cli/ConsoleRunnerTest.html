15  package de.aikiit.fotorenamer.cli;
16  
17  import de.aikiit.fotorenamer.util.Version;
18  import org.junit.jupiter.api.Test;
19  import org.junit.jupiter.api.io.TempDir;
20  import picocli.CommandLine;
21  
22  import javax.imageio.ImageIO;
23  import java.awt.image.BufferedImage;
24  import java.io.PrintWriter;
25  import java.io.StringWriter;
26  import java.nio.file.Files;
27  import java.nio.file.Path;
28  
29  import static org.assertj.core.api.Assertions.assertThat;
30  
31  class ConsoleRunnerTest {
32      @TempDir
33      Path directory;
34  
35      private final StringWriter out = new StringWriter();
36      private final StringWriter err = new StringWriter();
37  
38      private int execute(String... args) {
39          return new CommandLine(new ConsoleRunner())
40                  .setOut(new PrintWriter(out, true))
41                  .setErr(new PrintWriter(err, true))
42                  .execute(args);
43      }
44  
45      @Test
46      void helpAndVersionDoNotRequireDirectoryOrProcessImages() throws Exception {
47          Path image = copyImage("holiday.jpg");
48          assertThat(execute("--help")).isEqualTo(0);
49          assertThat(out.toString()).contains("DIRECTORY", "--reverse");
50          assertThat(execute("--version")).isEqualTo(0);
51          assertThat(out.toString()).contains(Version.VERSION);
52          assertThat(Files.exists(image)).isTrue();
53          assertThat(err.toString()).isEmpty();
54      }
55  
56      @Test
57      void rejectsMissingDirectoryAndUnknownOptions() {
58          assertThat(execute()).isEqualTo(2);
59          assertThat(err.toString()).contains("DIRECTORY");
60          assertThat(execute("--unknown", directory.toString())).isEqualTo(2);
61          assertThat(err.toString()).contains("Unknown option");
62          assertThat(execute(directory.toString(), "extra")).isEqualTo(2);
63      }
64  
65      @Test
66      void rejectsNonDirectoriesWithoutProcessingImages() throws Exception {
67          Path image = copyImage("holiday.jpg");
68          assertThat(execute(directory.resolve("missing").toString())).isEqualTo(2);
69          assertThat(execute(image.toString())).isEqualTo(2);
70          assertThat(err.toString()).contains("Not a directory");
71          assertThat(Files.exists(image)).isTrue();
72      }
73  
74      @Test
75      void reportsEmptyDirectoryAsFailure() {
76          assertThat(execute(directory.toString())).isEqualTo(1);
77          assertThat(err.toString()).contains("No images found");
78      }
79  
80      @Test
81      void renamesAndSkipsExactExifPrefixesInMixedBatch() throws Exception {
82          Path source = copyImage("holiday.dat");
83          byte[] content = Files.readAllBytes(source);
84          Path prefixed = copyImage("20110130_131102_existing.jpg");
85  
86          assertThat(execute(directory.toString())).isEqualTo(0);
87          assertThat(out.toString()).contains("Processed: 2 / 2", "Renamed: 1", "Skipped: 1", "Failed: 0");
88          assertThat(Files.exists(source)).isFalse();
89          assertThat(Files.readAllBytes(directory.resolve("20110130_131102_holiday.dat"))).containsExactly(content);
90          assertThat(Files.readAllBytes(prefixed)).containsExactly(content);
91      }
92  
93      @Test
94      void supportsBothReverseOptionsAndLegacyPrefixes() throws Exception {
95          for (String option : new String[]{"-r", "--reverse"}) {
96              Path source = copyImage("20110130_1311_holiday.jpg");
97              byte[] content = Files.readAllBytes(source);
98              assertThat(execute(option, directory.toString())).isEqualTo(0);
99              assertThat(Files.exists(source)).isFalse();
100             assertThat(Files.readAllBytes(directory.resolve("holiday.jpg"))).containsExactly(content);
101             Files.delete(directory.resolve("holiday.jpg"));
102         }
103     }
104 
105     @Test
106     void countsImagesWithoutExifAsSkipped() throws Exception {
107         Path image = directory.resolve("plain.png");
108         assertThat(ImageIO.write(new BufferedImage(2, 2, BufferedImage.TYPE_INT_RGB), "png", image.toFile())).isTrue();
109         byte[] content = Files.readAllBytes(image);
110         assertThat(execute(directory.toString())).isEqualTo(0);
111         assertThat(out.toString()).contains("Skipped: 1");
112         assertThat(Files.readAllBytes(image)).containsExactly(content);
113     }
114 
115     @Test
116     void reportsFailureWithoutOverwritingTargetsAndContinues() throws Exception {
117         Path source = copyImage("holiday.jpg");
118         byte[] content = Files.readAllBytes(source);
119         Path target = Files.writeString(directory.resolve("20110130_131102_holiday.jpg"), "keep this");
120         copyImage("other.jpg");
121         assertThat(execute(directory.toString())).isEqualTo(1);
122         assertThat(out.toString()).contains("Renamed: 1", "Failed: 1");
123         assertThat(Files.readAllBytes(source)).containsExactly(content);
124         assertThat(Files.readString(target)).isEqualTo("keep this");
125         assertThat(Files.exists(directory.resolve("20110130_131102_other.jpg"))).isTrue();
126     }
127 
128     private Path copyImage(String name) throws Exception {
129         Path target = directory.resolve(name);
130         try (var input = getClass().getResourceAsStream("/data/IMG_7559_mini.JPG")) {
131             assertThat(input).isNotNull();
132             Files.copy(input, target);
133         }
134         return target;
135     }
136 }
